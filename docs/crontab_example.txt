# 查看定时任务日志：
# - 清理日志: tail -f logs/cleanup_exports.log
# - 系统日志: grep CRON /var/log/syslog

# ------------------------------------------------------------------------------
# 每日清理任务：清理7天前的导出文件
# ------------------------------------------------------------------------------
# 执行时间：每天凌晨2点
# 功能：删除超过7天的CSV导出文件，释放磁盘空间
# 日志：输出到 logs/cleanup_exports.log
# ------------------------------------------------------------------------------
0 2 * * * cd /home/ubuntu/.openclaw/workspace/expiry-clean && php scripts/cleanup_exports.php >> logs/cleanup_exports.log 2>&1

# ------------------------------------------------------------------------------
# 每周深度清理：清理所有导出文件
# ------------------------------------------------------------------------------
# 执行时间：每周日凌晨3点
# 功能：删除所有CSV导出文件（包括7天内的）
# 用途：彻底清理，防止文件积累
# 注意：使用--full参数会忽略7天保留期
# ------------------------------------------------------------------------------
0 3 * * 0 cd /home/ubuntu/.openclaw/workspace/expiry-clean && php scripts/cleanup_exports.php --full >> logs/full_cleanup.log 2>&1

# ------------------------------------------------------------------------------
# Crontab时间格式说明
# ------------------------------------------------------------------------------
# * * * * * 命令
# | | | | |
# | | | | +---- 星期几 (0-7, 0和7都代表周日)
# | | | +------ 月份 (1-12)
# | | +-------- 日期 (1-31)
# | +---------- 小时 (0-23)
# +------------ 分钟 (0-59)
#
# 示例：
# 0 2 * * *     - 每天2点执行
# */5 * * * *   - 每5分钟执行
# 0 */2 * * *   - 每2小时执行
# 0 0 * * 0     - 每周日0点执行
# 0 0 1 * *     - 每月1号0点执行
# ------------------------------------------------------------------------------
